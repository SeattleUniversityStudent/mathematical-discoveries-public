def carlitz : Nat → Nat → Nat :=
fun n k => (List.filter adjDistinct (comps k n)).length
def comps : Nat → Nat → List (List Nat) :=
fun x x_1 => Nat.brecOn (motive := fun x => Nat → List (List Nat)) x comps._f x_1
def adjDistinct : List Nat → Bool :=
fun x => List.brecOn x adjDistinct._f
